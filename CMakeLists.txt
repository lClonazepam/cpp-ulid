cmake_minimum_required(VERSION 3.16)
project(cpp_ulid LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(test_ulid tests/test_ulid.cpp)
target_include_directories(test_ulid PRIVATE include)
if(MSVC)
  target_compile_options(test_ulid PRIVATE /W4)
else()
  target_compile_options(test_ulid PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME test_ulid COMMAND test_ulid)
